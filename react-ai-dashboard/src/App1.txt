import { useState } from 'react';
import './App.css';

function App() {
  const [query, setQuery] = useState('');
  const [file, setFile] = useState(null);
  const [schema, setSchema] = useState('');
  const [previewUrl, setPreviewUrl] = useState('');
  
  // States to handle Orchestrator responses
  const [orchestratorData, setOrchestratorData] = useState(null);
  const [loading, setLoading] = useState(false);

  const handleFileChange = (e) => {
    const selectedFile = e.target.files[0];
    setFile(selectedFile);
    setPreviewUrl(URL.createObjectURL(selectedFile));
  };

  const handleExecute = async () => {
    if (!query) {
      alert("Please enter a command or query first!");
      return;
    }

    setLoading(true);
    setOrchestratorData(null);

    // .NET API expects Multipart FormData
    const formData = new FormData();
    formData.append('userQuery', query);
    if (file) formData.append('file', file);
    if (schema) formData.append('schema', schema);

    try {
      // ⚠️ Note: check your exact local .NET port inside your terminal launchSettings.json
      const response = await fetch('http://localhost:5137/api/orchestrator/run', {
        method: 'POST',
        body: formData,
      });

      if (!response.ok) throw new Error("Orchestrator failed execution");

      const result = await response.json();
      setOrchestratorData(result);
    } catch (error) {
      console.error(error);
      alert("Could not communicate with the .NET Orchestrator Core.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="main-wrapper" style={{ maxWidth: '900px', margin: '40px auto', fontFamily: '"Segoe UI", sans-serif', padding: '0 20px' }}>
      
      <header style={{ borderBottom: '2px solid #eaeaea', paddingBottom: '20px', marginBottom: '30px' }}>
        <h2 style={{ color: '#222', margin: 0 }}>M4 Intelligent AI Orchestrator 🧠</h2>
        <p style={{ color: '#666', marginTop: '5px' }}>React UI → .NET 8 Core Router → Local Models (Ollama & Python)</p>
      </header>

      {/* Control Console */}
      <div style={{ background: '#f8f9fa', padding: '25px', borderRadius: '12px', boxShadow: '0 4px 6px rgba(0,0,0,0.05)' }}>
        <div style={{ marginBottom: '20px' }}>
          <label style={{ display: 'block', fontWeight: 'bold', marginBottom: '8px', textAlign: 'left' }}>Universal Prompt / Query:</label>
          <input 
            type="text" 
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            placeholder="E.g., 'Edit this image to make it winter' OR 'Draw a glowing neon cat' OR 'Fetch marketing employees'..."
            style={{ width: '100%', padding: '12px', fontSize: '16px', borderRadius: '6px', border: '1px solid #ccc', boxSizing: 'border-box' }}
          />
        </div>

        <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '20px', marginBottom: '20px' }}>
          <div style={{ textAlign: 'left' }}>
            <label style={{ display: 'block', fontWeight: 'bold', marginBottom: '8px' }}>Attach Photo (For Image Editing):</label>
            <input type="file" accept="image/*" onChange={handleFileChange} style={{ fontSize: '14px' }} />
            {previewUrl && (
              <img src={previewUrl} alt="Original Attachment" style={{ maxWidth: '120px', marginTop: '10px', borderRadius: '6px', display: 'block', border: '1px solid #ddd' }} />
            )}
          </div>

          <div style={{ textAlign: 'left' }}>
            <label style={{ display: 'block', fontWeight: 'bold', marginBottom: '8px' }}>DB Schema Context (For Text-to-SQL):</label>
            <textarea 
              value={schema}
              onChange={(e) => setSchema(e.target.value)}
              placeholder="Table: Employees (Id INT, Name VARCHAR, Role VARCHAR)"
              rows={3}
              style={{ width: '100%', padding: '10px', borderRadius: '6px', border: '1px solid #ccc', fontFamily: 'monospace' }}
            />
          </div>
        </div>

        <button 
          onClick={handleExecute}
          disabled={loading}
          style={{ width: '100%', padding: '14px', background: '#0062cc', color: 'white', border: 'none', borderRadius: '6px', fontSize: '16px', fontWeight: 'bold', cursor: 'pointer', transition: 'background 0.2s' }}
        >
          {loading ? "⚙️ Orchestrating Workflow & Processing..." : "Process Request 🚀"}
        </button>
      </div>

      {/* Dynamic Response Viewer */}
      {orchestratorData && (
        <div style={{ marginTop: '40px', textAlign: 'left', background: '#ffffff', border: '1px solid #e1e4e6', borderRadius: '12px', padding: '25px', boxShadow: '0 4px 12px rgba(0,0,0,0.05)' }}>
          <h3 style={{ margin: '0 0 15px 0', borderBottom: '1px solid #eee', paddingBottom: '10px' }}>
            Engine Executed: <span style={{ color: '#28a745' }}>{orchestratorData.actionPerformed}</span>
          </h3>

          {/* Condition 1: Image Processing Output */}
          {orchestratorData.dataType === 'IMAGE_BASE64' && (
            <div style={{ textAlign: 'center' }}>
              <p style={{ color: '#555', fontWeight: '500', marginBottom: '15px' }}>Processed Output Image:</p>
              <img 
                src={orchestratorData.content} 
                alt="AI Output Result" 
                style={{ maxWidth: '100%', maxHeight: '450px', borderRadius: '8px', boxShadow: '0 10px 25px rgba(0,0,0,0.15)', border: '2px solid #28a745' }} 
              />
            </div>
          )}

          {/* Condition 2: Text / Web Search Output */}
          {orchestratorData.dataType === 'TEXT' && (
            <div style={{ background: '#f1f3f5', padding: '20px', borderRadius: '8px', borderLeft: '4px solid #0062cc' }}>
              <p style={{ margin: 0, fontSize: '15px', color: '#333', lineHeight: '1.6' }}>{orchestratorData.content}</p>
            </div>
          )}

          {/* Condition 3: Database Query & Data JSON Output */}
          {orchestratorData.dataType === 'JSON_TABLE' && (
            <div>
              <div style={{ background: '#212529', color: '#a5d6a7', padding: '12px 18px', borderRadius: '6px', fontFamily: 'monospace', marginBottom: '15px' }}>
                <strong>Generated SQL Query:</strong> <span style={{ color: '#fff' }}>{orchestratorData.content.query}</span>
              </div>
              <div style={{ background: '#f8f9fa', padding: '15px', borderRadius: '6px', border: '1px solid #ddd' }}>
                <strong style={{ display: 'block', marginBottom: '8px' }}>Fetched Rows:</strong>
                <pre style={{ margin: 0, fontSize: '13px', overflowX: 'auto' }}>
                  {JSON.stringify(orchestratorData.content.rows, null, 2)}
                </pre>
              </div>
            </div>
          )}
        </div>
      )}
    </div>
  );
}

export default App;
