import ssl

ssl._create_default_https_context = ssl._create_unverified_context

# Hugging Face hub ke liye SSL verify false set karna
os.environ["HF_HUB_DISABLE_SYMLINKS_WARNING"] = "1"

from fastapi import FastAPI, UploadFile, File, Form
from fastapi.responses import FileResponse
import torch
# StableDiffusionInstructionPix2PixPipeline ki jagah StableDiffusionInstructPix2PixPipeline use hota hai naye versions mein
# Par safer side ke liye hum common DiffusionPipeline use karenge
from diffusers import DiffusionPipeline, StableDiffusionPipeline
from PIL import Image
import io

app = FastAPI(title="M4 Local AI Service")

# 1. Mac M4 (Metal) Core Graphic System Check
device = "mps" if torch.backends.mps.is_available() else "cpu"
print(f"Using Device: {device.upper()} for AI Generation")

# 2. AI Models Loading
print("Loading Image Editing Model (InstructPix2Pix)...")
# Generic DiffusionPipeline standard checkpoint ko sahi se wrap kar leta hai
edit_pipe = DiffusionPipeline.from_pretrained(
    "timbrooks/instruct-pix2pix", 
    torch_dtype=torch.float16 if device == "mps" else torch.float32, 
    safety_checker=None
).to(device)

print("Loading Image Generation Model (Stable Diffusion 1.5)...")
create_pipe = StableDiffusionPipeline.from_pretrained(
    "runwayml/stable-diffusion-v1-5", 
    torch_dtype=torch.float16 if device == "mps" else torch.float32,
    safety_checker=None
).to(device)

print("⚡ All Local Models Loaded Successfully on Mac M4 Neural Core!")

# --- ENDPOINTS ---

@app.post("/process")
async def edit_image(file: UploadFile = File(...), prompt: str = Form(...)):
    contents = await file.read()
    orig_image = Image.open(io.BytesIO(contents)).convert("RGB")
    
    # ⚡ FIX 1: Aspect Ratio distortion se bachne ke liye Square Padding logic
    max_side = max(orig_image.size)
    padded_image = Image.new("RGB", (max_side, max_side), (255, 255, 255))
    # Original image ko center me paste karna bina stretch kiye
    offset = ((max_side - orig_image.size[0]) // 2, (max_side - orig_image.size[1]) // 2)
    padded_image.paste(orig_image, offset)
    
    # Ab bina distorion ke perfect 512x512 par resize hoga
    input_image = padded_image.resize((512, 512), Image.Resampling.LANCZOS)

    print(f"[EDIT REQUEST] Prompt: {prompt}")
    
    # ⚡ FIX 2: Parameters ko adjust kiya taaki original face structure intact rahe
    edited_image = edit_pipe(
        prompt, 
        image=input_image, 
        num_inference_steps=30,      # Steps 30 rakhein taaki details crisp hon
        image_guidance_scale=2.0,    # 🔥 ISKO 1.8 se badhakar 2.2 ya 2.5 tak karein agar face badal raha hai
        guidance_scale=9.0           # Text command control
    ).images[0]
    
    # ⚡ FIX 3: Output ko wapas original landscape/portrait layout me crop karna
    # Taaki final response horizontal ya vertically khicha hua na lage
    ratio = max_side / 512
    crop_box = (
        int(offset[0] / ratio),
        int(offset[1] / ratio),
        int((offset[0] + orig_image.size[0]) / ratio),
        int((offset[1] + orig_image.size[1]) / ratio)
    )
    final_output = edited_image.crop(crop_box).resize(orig_image.size, Image.Resampling.LANCZOS)
    
    output_path = "edit_result.jpg"
    final_output.save(output_path, quality=95)
    return FileResponse(output_path, media_type="image/jpeg")



@app.post("/generate")
async def generate_image(prompt: str = Form(...)):
    print(f"[CREATE REQUEST] Prompt: {prompt}")
    
    # ⚡ FIX 1: Quality aur Text-Alignment Controls
    # Steps ko 25 kiya taaki M4 GPU par sharp details aur textures generate hon
    # guidance_scale=7.5 add kiya taaki model aapke text prompt ko accurate follow kare
    generated_image = create_pipe(
        prompt, 
        num_inference_steps=25,
        guidance_scale=7.5,
        width=512,
        height=512
    ).images[0]
    
    # ⚡ FIX 2: Lanczos Resampling for smooth pixel artifacts
    final_output = generated_image.resize((512, 512), Image.Resampling.LANCZOS)
    
    output_path = "create_result.jpg"
    # ⚡ FIX 3: Higher quality compression toggle (95% clear textures)
    final_output.save(output_path, quality=95)
    
    # Aapka exact original responsive return format return ho raha hai
    return FileResponse(output_path, media_type="image/jpeg")
